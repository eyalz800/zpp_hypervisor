# Bochs configuration for debugging zpp_hypervisor under emulated VT-x.
# Paths are relative, so run Bochs from build/bochs.
plugin_ctrl: unmapped=1, biosdev=1, speaker=1, extfpuirq=1, parallel=1, serial=1
config_interface: textconfig
display_library: nogui

# OVMF is a 4 MB flash image, so it maps at 4 GB - 4 MB.
romimage: file=OVMF.fd, address=0xffc00000, options=fastboot
memory: guest=1024, host=1024

# A model that reports VMX. Bochs 3.0 removed the legacy 'cpuid:' option,
# so CPU features come from the model alone - pick one with VT-x.
# ips is Bochs' estimate of how fast it can simulate, and it drives the guest's
# sense of time rather than actual speed. Too low a value makes firmware and
# guests hit their own internal timeouts, so keep it realistic.
#
# One processor is a default here, not a restriction. Raise it when the
# multiprocessor paths are what is being debugged - the start-up IPI half of
# the self check has nothing to send an IPI to at count=1.
#
# It used to say the stub and SMP could not coexist, on the strength of the
# #error in bochs.h. That #error is real but is only a guard: its own comment
# says the problem is knowing which CPU gdb refers to, and pinning the stub to
# processor 0 answers that. Built and run with both enabled, four processors
# reach ZPP_HYPERVISOR_ACTIVE with gdb attached. The surviving limitation is
# the one the guard was about - gdb only ever sees processor 0.
#
# scripts/ci/bochs-smoke.sh rewrites this line to $CPUS regardless.
cpu: model=corei7_icelake_u, count=1, ips=250000000, reset_on_triple_fault=1

ata0: enabled=1, ioaddr1=0x1f0, ioaddr2=0x3f0, irq=14
ata0-master: type=disk, mode=flat, path="esp.img"
boot: disk

# OVMF writes its console to the serial port.
com1: enabled=1, mode=file, dev=serial.out

log: bochs.log
logprefix: %t%e%d
panic: action=fatal
error: action=report
info: action=ignore
debug: action=ignore

# Bochs halts at reset and waits for gdb to connect on this port.
gdbstub: enabled=1, port=1337, text_base=0, data_base=0, bss_base=0
